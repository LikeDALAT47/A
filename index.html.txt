        /* ===================================================
           1. AUDIO SYNTHESIZER (WEB AUDIO API - NO EXTERNAL FILES)

        class SoundEngine {
            constructor() {
                this.ctx = null;
                this.isBgPlaying = false;
                this.rainNode = null;
            }

            init() {
                if (!this.ctx) {
                    const AudioContext = window.AudioContext || window.webkitAudioContext;
                    this.ctx = new AudioContext();
                }
                if (this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }

            // Synthesize Wooden Fish sound (Mõ)
            playWoodenFish(pitchModifier = 1) {
                this.init();
                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();

                // Pitch setup for hollow wood block effect
                osc.type = 'sine';
                osc.frequency.setValueAtTime(420 * pitchModifier, now);
                osc.frequency.exponentialRampToValueAtTime(100 * pitchModifier, now + 0.12);

                // Quick attack & fast exponential decay envelope
                gain.gain.setValueAtTime(1, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);

                osc.connect(gain);
                gain.connect(this.ctx.destination);

                osc.start(now);
                osc.stop(now + 0.13);
            }

            // Synthesize Temple Bell sound (Chuông)
            playBell() {
                this.init();
                const now = this.ctx.currentTime;
                
                // Harmonic frequencies for bell metallic resonance
                const freqs = [520, 1040, 1560, 2080];
                freqs.forEach((freq, idx) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();

                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(freq, now);

                    const decay = 2.5 / (idx + 1);
                    gain.gain.setValueAtTime(0.3 / (idx + 1), now);
                    gain.gain.exponentialRampToValueAtTime(0.0001, now + decay);

                    osc.connect(gain);
                    gain.connect(this.ctx.destination);

                    osc.start(now);
                    osc.stop(now + decay);
                });
            }

            // Synthesize Ambient Meditation Noise (Pink noise / Rain simulation)
            toggleBgAmbient() {
                this.init();
                if (this.isBgPlaying) {
                    if (this.rainNode) this.rainNode.disconnect();
                    this.isBgPlaying = false;
                    return false;
                } else {
                    const bufferSize = this.ctx.sampleRate * 2;
                    const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
                    const output = noiseBuffer.getChannelData(0);

                    let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
                    for (let i = 0; i < bufferSize; i++) {
                        let white = Math.random() * 2 - 1;
                        b0 = 0.99886 * b0 + white * 0.0555179;
                        b1 = 0.99332 * b1 + white * 0.0750759;
                        b2 = 0.96900 * b2 + white * 0.1538520;
                        b3 = 0.86650 * b3 + white * 0.3104856;
                        b4 = 0.55000 * b4 + white * 0.5329522;
                        b5 = -0.7616 * b5 - white * 0.0168980;
                        output[i] = b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362;
                        output[i] *= 0.03; // Gentle low volume rain
                        b6 = white * 0.115926;
                    }

                    const whiteNoise = this.ctx.createBufferSource();
                    whiteNoise.buffer = noiseBuffer;
                    whiteNoise.loop = true;

                    const filter = this.ctx.createBiquadFilter();
                    filter.type = 'lowpass';
                    filter.frequency.value = 800;

                    whiteNoise.connect(filter);
                    filter.connect(this.ctx.destination);

                    whiteNoise.start();
                    this.rainNode = whiteNoise;
                    this.isBgPlaying = true;
                    return true;
                }
            }
        }

        /* ===================================================

        const canvas = document.getElementById('particleCanvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Particle {
            constructor() {
                this.reset();
            }
            reset() {
                this.x = Math.random() * canvas.width;
                this.y = canvas.height + Math.random() * 20;
                this.size = Math.random() * 2 + 0.5;
                this.speedY = Math.random() * 0.6 + 0.2;
                this.opacity = Math.random() * 0.5 + 0.2;
            }
            update() {
                this.y -= this.speedY;
                if (this.y < -10) this.reset();
            }
            draw() {
                ctx.fillStyle = `rgba(223, 172, 68, ${this.opacity})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        for (let i = 0; i < 40; i++) {
            particles.push(new Particle());
        }

        function animateParticles() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.forEach(p => {
                p.update();
                p.draw();
            });
            requestAnimationFrame(animateParticles);
        }

        let autoTapInterval = null;
        let currentSkin = 'classic';

        // DOM Elements
        const personalCountEl = document.getElementById('personalCount');
        const globalMeritCountEl = document.getElementById('globalMeritCount');
        const liveUserCountEl = document.getElementById('liveUserCount');
        const woodenFishBtn = document.getElementById('woodenFishBtn');
        const woodenStick = document.getElementById('woodenStick');
        const auraEffect = document.getElementById('auraEffect');
        const fishBodyPath = document.getElementById('fishBodyPath');
        const autoTapBtn = document.getElementById('autoTapBtn');
        const autoTapText = document.getElementById('autoTapText');

        // Init Counts
        personalCountEl.innerText = personalMerit.toLocaleString();
        globalMeritCountEl.innerText = globalMerit.toLocaleString();

        // Simulated Global Live User Counter Ticker
        setInterval(() => {
            const randomAdd = Math.floor(Math.random() * 8) + 1;
            globalMerit += randomAdd;
            globalMeritCountEl.innerText = globalMerit.toLocaleString();

            // Slight fluctuation in live active users
            const liveUsers = 1048291 + Math.floor(Math.random() * 150) - 75;
            liveUserCountEl.innerHTML = `<i class="fa-solid fa-users text-[10px] mr-1"></i>${liveUsers.toLocaleString()} Phật tử`;
        }, 1500);

        // Core Tap Function
        function triggerTap(e) {
            personalMerit++;
            globalMerit++;
            localStorage.setItem('personalMerit', personalMerit);

            // Update UI Counters
            personalCountEl.innerText = personalMerit.toLocaleString();
            globalMeritCountEl.innerText = globalMerit.toLocaleString();

            // Play Audio
            const pitch = currentSkin === 'jade' ? 1.3 : (currentSkin === 'gold' ? 1.1 : 1.0);
            audioApp.playWoodenFish(pitch);

            // Visual Animations
            woodenFishBtn.classList.add('fish-tap');
            woodenStick.classList.add('stick-tap');
            auraEffect.style.opacity = '1';

            setTimeout(() => {
                woodenFishBtn.classList.remove('fish-tap');
                woodenStick.classList.remove('stick-tap');
                auraEffect.style.opacity = '0.6';
            }, 100);

            // Floating Blessing Text (+1 Công Đức)
            createFloatingText(e);
        }

        // Create Floating Text Effect
        function createFloatingText(e) {
            const textEl = document.createElement('div');
            textEl.className = 'float-text';
            textEl.innerText = `+1 ${selectedBlessing}`;

            // Coordinates calculation
            let x, y;
            if (e && e.clientX) {
                x = e.clientX - 40;
                y = e.clientY - 40;
            } else {
                const rect = woodenFishBtn.getBoundingClientRect();
                x = rect.left + rect.width / 2 - 30 + (Math.random() * 40 - 20);
                y = rect.top + 20;
            }

            textEl.style.left = `${x}px`;
            textEl.style.top = `${y}px`;

            if (currentSkin === 'gold') textEl.style.color = '#ffe885';
            else if (currentSkin === 'jade') textEl.style.color = '#a3f7bf';
            else textEl.style.color = '#dfac44';

            document.body.appendChild(textEl);
            setTimeout(() => textEl.remove(), 1200);
        }

        // Event Listeners for Tapping
        woodenFishBtn.addEventListener('pointerdown', (e) => {
            triggerTap(e);
        });

        // Blessing Tag Selector
        document.querySelectorAll('.blessing-tag').forEach(btn => {
            btn.addEventListener('click', function() {
                document.querySelectorAll('.blessing-tag').forEach(b => {
                    b.classList.remove('active', 'bg-zen-gold/20', 'border-zen-gold', 'text-zen-gold');
                    b.classList.add('bg-zen-card', 'border-zen-gold/20', 'text-zen-muted');
                });
                this.classList.add('active', 'bg-zen-gold/20', 'border-zen-gold', 'text-zen-gold');
                this.classList.remove('bg-zen-card', 'text-zen-muted');
                selectedBlessing = this.innerText.split(' ')[1];
            });
        });

        // Auto Tap Toggle
        autoTapBtn.addEventListener('click', () => {
            if (autoTapInterval) {
                clearInterval(autoTapInterval);
                autoTapInterval = null;
                autoTapText.innerText = 'Tự Động Gõ: TẮT';
                autoTapBtn.classList.remove('bg-zen-gold/20', 'border-zen-gold');
            } else {
                autoTapText.innerText = 'Tự Động Gõ: BẬT';
                autoTapBtn.classList.add('bg-zen-gold/20', 'border-zen-gold');
                autoTapInterval = setInterval(() => {
                    triggerTap(null);
                }, 800);
            }
        });

        // Bell Sound Button
        document.getElementById('bellBtn').addEventListener('click', () => {
            audioApp.playBell();
        });

        // Background Music Button
        const bgMusicBtn = document.getElementById('bgMusicBtn');
        const bgMusicText = document.getElementById('bgMusicText');
        bgMusicBtn.addEventListener('click', () => {
            const isPlaying = audioApp.toggleBgAmbient();
            if (isPlaying) {
                bgMusicBtn.classList.add('bg-zen-gold/20');
                bgMusicText.innerText = 'Tắt Nhạc';
            } else {
                bgMusicBtn.classList.remove('bg-zen-gold/20');
                bgMusicText.innerText = 'Nhạc Nhè Nhè';
            }
        });

        /* ===================================================
           4. MODALS & SKIN UNLOCK (MONETIZATION SYSTEM)
           =================================================== */
        const skinModal = document.getElementById('skinModal');
        const openSkinModalBtn = document.getElementById('openSkinModalBtn');
        const closeSkinModalBtn = document.getElementById('closeSkinModalBtn');

        openSkinModalBtn.addEventListener('click', () => skinModal.classList.remove('hidden'));
        closeSkinModalBtn.addEventListener('click', () => skinModal.classList.add('hidden'));

        // Skin change logic
        function applySkin(skinName) {
            currentSkin = skinName;
            if (skinName === 'classic') {
                fishBodyPath.setAttribute('fill', 'url(#skinClassic)');
            } else if (skinName === 'gold') {
                fishBodyPath.setAttribute('fill', 'url(#skinGold)');
            } else if (skinName === 'jade') {
                fishBodyPath.setAttribute('fill', 'url(#skinJade)');
            }
            skinModal.classList.add('hidden');
        }

        document.querySelectorAll('.select-skin-btn').forEach(btn => {
            btn.addEventListener('click', function() {
                applySkin(this.dataset.skin);
            });
        });

        // Simulated Rewarded Ad unlock flow
        document.querySelectorAll('.watch-ad-unlock-btn').forEach(btn => {
            btn.addEventListener('click', function() {
                const skinToUnlock = this.dataset.skin;
                const originalText = this.innerHTML;
                this.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> Đang xem Ads (5s)...`;
                this.disabled = true;

                setTimeout(() => {
                    alert('Cảm ơn bạn đã xem Quảng Cáo ủng hộ Chùa Online! Skin đã được mở khóa.');
                    applySkin(skinToUnlock);
                    this.innerHTML = 'Đã mở khóa';
                    this.classList.remove('bg-gradient-to-r');
                    this.classList.add('bg-zen-card', 'text-zen-gold');
                }, 3000);
            });
        });

        // Donation Modal logic
        const donationModal = document.getElementById('donationModal');
        document.getElementById('openDonationBtn').addEventListener('click', () => donationModal.classList.remove('hidden'));
        document.getElementById('closeDonationModalBtn').addEventListener('click', () => donationModal.classList.add('hidden'));
    </script>
