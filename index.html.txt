    <script>
        let count = parseInt(localStorage.getItem('my_merit')) || 0;
        document.getElementById('count').innerText = count;

        let globalCount = 1002450;
        setInterval(() => {
            globalCount += Math.floor(Math.random() * 5) + 1;
            document.getElementById('globalCount').innerText = globalCount.toLocaleString();
        }, 2000);

        const countEl = document.getElementById('count');
        const fishBtn = document.getElementById('fishBtn');
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound() {
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            
            osc.type = 'sine';
            osc.frequency.setValueAtTime(350, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(70, audioCtx.currentTime + 0.15);
            
            gain.gain.setValueAtTime(1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
            
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            osc.start();
            osc.stop(audioCtx.currentTime + 0.15);
        }

        fishBtn.addEventListener('click', (e) => {
            count++;
            countEl.innerText = count;
            localStorage.setItem('my_merit', count);
            
            playSound();

            const text = document.createElement('div');
            text.className = 'text-effect';
            text.innerText = '+1 Công Đức 🙏';
            text.style.left = (e.clientX - 20) + 'px';
            text.style.top = (e.clientY - 30) + 'px';
            document.body.appendChild(text);

            setTimeout(() => text.remove(), 800);
        });
    </script>
